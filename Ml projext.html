df = df.loc[:, ~df.columns.duplicated()]

df.drop_duplicates(inplace=True)

# 4. Rating & score columns ko numeric type me convert karna
numeric_cols = ['Customer_Reviews_Importance', 'Rating_Accuracy', 'Shopping_Satisfaction']
for col in numeric_cols:
    if col in df.columns:
        df[col] = pd.to_numeric(df[col], errors='coerce')

# 5. Missing values handle karna
df['Product_Search_Method'] = df['Product_Search_Method'].fillna('Unknown')
df.fillna(df.median(numeric_only=True), inplace=True)

# 6. Categorical text standardize karna
if 'Gender' in df.columns:
    df['Gender'] = df['Gender'].str.strip().str.title()

#Task 3
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler

# Rule-based profiling define karna
def assign_segment(row):
    freq = str(row.get('Purchase_Frequency', '')).lower()
    sat = row.get('Shopping_Satisfaction', 3)
    
    if ('few times a week' in freq or 'multiple' in freq) and sat >= 4:
        return 'Frequent Buyers'
    elif sat <= 2 or 'price' in str(row.get('Cart_Abandonment_Factors', '')).lower():
        return 'At-Risk Customers'
    else:
        return 'Occasional Shoppers'

df['Customer_Profile'] = df.apply(assign_segment, axis=1)
print("Profile Distribution:\n", df['Customer_Profile'].value_counts())

# K-Means Clustering (Machine Learning approach)
cluster_cols = ['Shopping_Satisfaction', 'Rating_Accuracy']
scaler = StandardScaler()
scaled_features = scaler.fit_transform(df[cluster_cols].dropna())

kmeans = KMeans(n_clusters=3, random_state=42)
df.loc[df[cluster_cols].dropna().index, 'Cluster'] = kmeans.fit_predict(scaled_features)

print("\nCluster Averages:\n", df.groupby('Cluster')[cluster_cols].mean())
